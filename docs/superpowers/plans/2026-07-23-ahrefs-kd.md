```ts
import { ArgumentError } from '@jackwener/opencli/errors';

export type KdRow = {
  keyword: string;
  country: string;
  kd: number;
};

export function normalizeKeyword(raw: unknown): string {
  const keyword = String(raw ?? '').trim();
  if (!keyword) throw new ArgumentError('keyword is required');
  return keyword;
}

export function normalizeCountry(raw: unknown): string {
  const country = String(raw ?? 'us').trim().toLowerCase();
  if (!/^[a-z]{2}$/.test(country)) {
    throw new ArgumentError(
      `invalid country "${raw}". Use a two-letter code like us, uk, de`,
    );
  }
  return country;
}

export function parseKd(raw: unknown): number {
  if (raw === null || raw === undefined || raw === '') {
    throw new ArgumentError('kd is required');
  }
  const n = typeof raw === 'number' ? raw : Number(String(raw).trim());
  if (!Number.isInteger(n) || n < 0 || n > 100) {
    throw new ArgumentError(`kd must be an integer 0-100 (got ${raw})`);
  }
  return n;
}

export function toRows(keyword: string, country: string, kd: number): KdRow[] {
  return [{ keyword, country, kd: parseKd(kd) }];
}
```

async function fetchKd(keyword: string, country: string): Promise<number> {
  const { url, init } = buildKdRequest(keyword, country);
  let resp: Response;
  try {
    resp = await fetch(url, init);
  } catch (e) {
    throw new CliError(
      'FETCH_ERROR',
      `Network error: ${e instanceof Error ? e.message : String(e)}`,
      'Check your network connection',
    );
  }
  if (resp.status === 429) {
    throw new CommandExecutionError(
      'Ahrefs rate-limited the free Keyword Difficulty Checker (429). Retry later.',
    );
  }
  if (resp.status === 401 || resp.status === 403) {
    throw new CommandExecutionError(
      'Ahrefs free KD endpoint requires auth unexpectedly. Page may have changed.',
    );
  }
  if (!resp.ok) {
    throw new CommandExecutionError(`Ahrefs KD HTTP ${resp.status}`);
  }
  const text = await resp.text();
  let payload: unknown;
  try {
    payload = JSON.parse(text);
  } catch {
    throw new CommandExecutionError('Failed to parse Ahrefs KD response as JSON');
  }
  const raw = extractKdFromPayload(payload);
  if (raw === undefined || raw === null || raw === '') {
    throw new EmptyResultError('ahrefs kd', `No KD returned for "${keyword}" (${country})`);
  }
  try {
    return parseKd(raw);
  } catch (e) {
    if (e instanceof ArgumentError) {
      throw new EmptyResultError('ahrefs kd', `Invalid KD value for "${keyword}": ${raw}`);
    }
    throw e;
  }
